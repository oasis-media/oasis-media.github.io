<script>
  document.querySelectorAll<HTMLElement>(".bookmark-carousel").forEach((carousel) => {
    if (carousel.dataset.smoothScroll === "true") return;
    carousel.dataset.smoothScroll = "true";

    let target = carousel.scrollLeft;
    let animationFrame = 0;

    function animateScroll() {
      const distance = target - carousel.scrollLeft;

      if (Math.abs(distance) < 0.5) {
        carousel.scrollLeft = target;
        animationFrame = 0;
        return;
      }

      carousel.scrollLeft += distance * 0.15;
      animationFrame = requestAnimationFrame(animateScroll);
    }

    carousel.addEventListener("wheel", (event: WheelEvent) => {
      if (carousel.scrollWidth <= carousel.clientWidth) return;

      if (Math.abs(event.deltaY) > Math.abs(event.deltaX)) {
        event.preventDefault();

        target += event.deltaY;
        target = Math.max(
          0,
          Math.min(target, carousel.scrollWidth - carousel.clientWidth)
        );

        if (!animationFrame) {
          animationFrame = requestAnimationFrame(animateScroll);
        }
      }
    }, { passive: false });
  });
  
  document.querySelectorAll<HTMLElement>(".carousel-wrapper").forEach((wrapper) => {
    const carousel = wrapper.querySelector<HTMLElement>(".bookmark-carousel");
    const progress = wrapper.querySelector<HTMLElement>(".carousel-progress");
    const fill = wrapper.querySelector<HTMLElement>(".carousel-progress-fill");
    const prev = wrapper.querySelector<HTMLButtonElement>(".carousel-prev");
    const next = wrapper.querySelector<HTMLButtonElement>(".carousel-next");

    if (!carousel || !progress || !fill || !prev || !next) return;
    if (wrapper.dataset.controlsReady === "true") return;
    wrapper.dataset.controlsReady = "true";

    function updateProgress() {
      const maxScroll = carousel.scrollWidth - carousel.clientWidth;
      const percentage = maxScroll > 0
        ? (carousel.scrollLeft / maxScroll) * 100
        : 0;

      fill.style.width = `${percentage}%`;
      progress.setAttribute("aria-valuenow", String(Math.round(percentage)));

      prev.disabled = carousel.scrollLeft <= 1;
      next.disabled = carousel.scrollLeft >= maxScroll - 1;
    }

    function scrollByAmount(direction: number) {
      carousel.scrollBy({
        left: direction * carousel.clientWidth * 0.8,
        behavior: "smooth",
      });
    }

    prev.addEventListener("click", () => scrollByAmount(-1));
    next.addEventListener("click", () => scrollByAmount(1));

    progress.addEventListener("click", (event) => {
      const rect = progress.getBoundingClientRect();
      const fraction = Math.max(0, Math.min(1, (event.clientX - rect.left) / rect.width));
      const maxScroll = carousel.scrollWidth - carousel.clientWidth;

      carousel.scrollTo({
        left: fraction * maxScroll,
        behavior: "smooth",
      });
    });

    progress.addEventListener("keydown", (event) => {
      if (event.key === "ArrowRight") {
        event.preventDefault();
        scrollByAmount(1);
      } else if (event.key === "ArrowLeft") {
        event.preventDefault();
        scrollByAmount(-1);
      } else if (event.key === "Home" || event.key === "End") {
        event.preventDefault();
        carousel.scrollTo({
          left: event.key === "Home" ? 0 : carousel.scrollWidth,
          behavior: "smooth",
        });
      }
    });

    carousel.addEventListener("scroll", updateProgress, { passive: true });
    window.addEventListener("resize", updateProgress);
    updateProgress();
  });
</script>
